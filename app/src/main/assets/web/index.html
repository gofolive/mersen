2. PostgreSQL Table DDL (Relational Database):

CREATE TABLE students ( id SERIAL PRIMARY KEY, mtu_id VARCHAR(30) UNIQUE NOT NULL, full_name VARCHAR(100) NOT NULL, department VARCHAR(80) NOT NULL, year_of_study INT NOT NULL, dietary_type VARCHAR(30) NOT NULL, dining_hall VARCHAR(30) NOT NULL, status VARCHAR(20) DEFAULT 'ACTIVE' ); CREATE TABLE meal_scan_logs ( id SERIAL PRIMARY KEY, scan_id VARCHAR(36) UNIQUE NOT NULL, student_id VARCHAR(30) REFERENCES students(mtu_id), meal_session VARCHAR(20) NOT NULL, timestamp TIMESTAMP WITH TIME ZONE DEFAULT NOW(), dining_hall VARCHAR(30) NOT NULL, status VARCHAR(30) NOT NULL, is_offline_cached BOOLEAN DEFAULT FALSE );
